Task 1 — Custom Array Transformation
Problem: Implement a function transformArray that takes an array and a callback function. Apply the callback to every element and return a new array.

solution:
function transformArray(arr,callback){
  let res=[];
  for(let i=0;i<arr.length;i++){
    res.push(callback(arr[i]));
  }
  return res;
}
let arr=[1,2,3,4];
let ans=transformArray([1, 2, 3, 4], x => x * 2);
console.log(ans);

Task 2 — Filter Employees by Salary
Problem: Given an array of employees, implement filterEmployees that accepts a callback to determine which employees should be returned.

solution:
const employees = [
    { name: "Asha", salary: 50000 },
    { name: "Ravi", salary: 80000 },
    { name: "Priya", salary: 65000 },
    { name: "Kiran", salary: 40000 }
];
function filterEmployees(employees,callback){
  let res=[];
  for(let i=0;i<employees.length;i++){
    if(callback(employees[i])){
       res.push(employees[i]);
    }
  }
  return res;
}
let ans=filterEmployees(employees, emp => emp.salary >= 60000);
console.log(ans);

function calculate(a, b, operation) {

Task 5 — Process Orders Using Multiple Callbacks
Problem: Create processOrders(orders, validator, discountCalculator).

Each order has an amount and a status:

solution:
const orders = [
    { id: 1, amount: 1000, status: "paid" },
    { id: 2, amount: 500, status: "pending" },
    { id: 3, amount: 2000, status: "paid" }
];

const validator = order => order.status === "paid";

const discountCalculator = order => order.amount * 0.9;

function processOrders(orders, validator, discountCalculator) {
    let result = [];

    for (let i = 0; i < orders.length; i++) {
        if (validator(orders[i])) {
            let discountedAmount = discountCalculator(orders[i]);

            result.push({
                ...orders[i],
                finalAmount: discountedAmount
            });
        }
    }

    return result;
}

let ans = processOrders(orders, validator, discountCalculator);

console.log(ans);

Task 7 — Bank Account With Private Balance
Problem: Implement createBankAccount(initialBalance) that returns an object with three methods: deposit(amount), withdraw(amount), and getBalance().

solution:
function createBankAccount(balance){
  function deposit(amount){
    balance+=amount;
  }
  function withdraw(amount){
    balance-=amount;
  }
  function getBalance(){
    return balance;
  }
  return{
    deposit,
    withdraw,
    getBalance
  };
}

const account = createBankAccount(1000);

account.deposit(500);
console.log(account.getBalance()); // 1500

account.withdraw(200);
console.log(account.getBalance()); // 1300

Task 9 — Create a Function That Remembers Its Previous Result
Problem: Implement createRememberingFunction(fn) that remembers the most recent input and its result. If the next call uses the same input,
return the cached result without calling fn again.

// here for this task we need to know about the caching 

What is caching in JavaScript?
Caching is the technique of storing the result of an operation so that when the same operation is needed again, we can reuse the stored result instead of calculating or fetching it again.
Think of it like this:
Imagine you look up the capital of India. The first time, you search for it and learn that it is New Delhi. The next time someone asks you, you already know the answer, 
so you don't need to search again.
That is the basic idea of caching.

--->Without caching vs. with caching
--->Without caching
Every request repeats the calculation or fetch.
More repeated work

---->With caching
Reuse a previously stored result when possible.
Less repeated work

let calls=0;
function square(n) {
    calls++;
    return n * n;
}

function  createRememberingFunction(fn) {
  
  let cache={};
  return function(n){
     if (Object.hasOwn(cache, n)) {
            console.log("Getting result from cache");
            return cache[n];
        }
  
  console.log("Calculating...");
        const result = fn(n);
        cache[n] = result;
        return result;
  };
}

const rememberedSquare = createRememberingFunction(square);

console.log(rememberedSquare(5)); // 25
console.log(rememberedSquare(5)); // 25
console.log(calls);               // 1

console.log(rememberedSquare(6)); // 36
console.log(calls);               // 2

Task 13 — Implement once()
Problem: Implement once(fn) so that the wrapped function executes only on its first call. Subsequent calls return the first result without executing fn again.

function add(a, b) {
    console.log("Function executed");
    return a+b;
}

function once(fn) {
    let executed=false;
    let res;
    return function(...args) {
          if(executed===false){
            res=fn(...args);
            executed=true;
          }
          return res;
    };
}

const onceAdd = once(add);

console.log(onceAdd(2, 3));
console.log(onceAdd(10, 20));
console.log(onceAdd(5, 5));
